\section{RESEARCH EXPERIENCE}
% ============================================================

\entry{Graduate Research Assistant}{Jan 2021 -- Present}
      {University of Massachusetts Amherst}{Amherst, MA}

\vspace{4pt}
\textbf{Schematic Bicycle Map Design and Micromobility Behavior}
\begin{itemize}
  \item Mapped the national landscape of schematic bicycle map adoption across 684 US urban counties; Bayesian mixture modeling with XGBoost and SHAP analysis revealed three county types of adoption.
  \item Designed and deployed an incentivized national survey on how cyclists and micromobility riders respond to map design: simplification level, geographic accuracy, infrastructure representation, and route legibility.
  \item Modeled stated intentions to ride under different map designs with a Bayesian latent class ordered logit that separates riders into groups with different design sensitivities.
  \item Formulated a mixed-integer linear program that selects bicycle map routes from street and trail networks under safety, connectivity, and user-preference constraints.
  \item Built a bot detection pipeline for online surveys that combines platform signals, browser fingerprinting, timing checks, and an XGBoost validation model.
  \item \textit{Funding: Armstrong Fund for Science, 2024--2026 (PI: J.~Oke, co-PI: E.~Christofa); NSF REU, 2023--2026.}
  \item \textit{2 journal articles published, 3 under review, 2 in preparation.}
\end{itemize}

\vspace{4pt}
\textbf{Spatial Crash Typology and Risk Prediction}
\begin{itemize}
  \item Classified 2,480 New England census tracts into crash types with UMAP and Gaussian mixture models, using one year of state-level crash records.
  \item Linked patterns in crash frequency and severity to roadway geometry, land use, and socioeconomic conditions across urban, suburban, and rural tracts.
  \item Trained XGBoost models with SHAP interpretation to predict crash risk within each type and show which factors drive risk where, informing where safety interventions should go.
  \item \textit{Funding: New England University Transportation Center (NEUTC), 2024--2025 (PI: J.~Oke).}
  \item \textit{1 journal article published.}
\end{itemize}

\vspace{4pt}
\textbf{Transit OD Inference from Partial Mobile Ticketing Data}
\begin{itemize}
  \item Built a framework that infers full origin-destination-transfer matrices from boarding-only mobile ticketing data on a regional bus network with 40+ routes.
  \item Clustered 40,000+ weekly riders into passenger types using hierarchical clustering with dynamic time warping.
  \item Added a gradient boosting spatial error correction step that cut MAE by 70\% and SMAPE by 85\% relative to the baseline trip chaining.
  \item \textit{Funding: Pioneer Valley Transit Authority (PVTA), 2021--2023 (PI: J.~Oke).}
  \item \textit{3 journal articles published, 1 in preparation.}  
\end{itemize}
