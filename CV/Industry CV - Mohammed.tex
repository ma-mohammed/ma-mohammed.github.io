\section{RESEARCH \& PROJECT EXPERIENCE}
% ============================================================

\entry{Graduate Research Assistant}{Jan 2021 -- Present}
      {University of Massachusetts Amherst}{Amherst, MA, USA}

\vspace{4pt}
\textbf{Transit OD Inference from Mobile Ticketing Data (PVTA)}
\begin{itemize}
  \item Built an ML pipeline to infer complete origin-destination-transfer matrices from boarding-only mobile ticketing records across a 40+ route regional bus network, reducing MAE by 70\% and SMAPE by 85\% over baseline trip-chaining models.
  \item Modeled passenger behavioral heterogeneity across 40,000+ weekly riders using hierarchical clustering and dynamic time warping, producing seasonal rider typologies.
  \item Deployed a public-facing interactive OD dashboard for PVTA (\href{https://pvtaodx.narslab.org}{pvtaodx.narslab.org}), built with Plotly Dash, containerized and hosted on AWS Lightsail.
\end{itemize}

\vspace{4pt}
\textbf{Spatial Crash Typology and Risk Prediction (New England)}
\begin{itemize}
  \item Integrated crash records, roadway geometry, land use, and socioeconomic attributes for 2,480 New England census tracts. Applied UMAP and Gaussian mixture modeling to classify tracts into safety typologies.
  \item Developed XGBoost predictive models with SHAP-based interpretation to forecast crash risk by typology and identify high-impact features for targeted safety interventions.
\end{itemize}

\vspace{4pt}
\textbf{Schematic Bicycle Map Design and Micromobility Behavior}
\begin{itemize}
  \item Analyzed schematic bicycle map adoption patterns across 684 US urban counties using Bayesian mixture modeling, XGBoost, and SHAP, producing three county typologies to inform national map design guidance.
  \item Built a MILP optimization framework to select optimal bicycle network routes for schematic mapping, balancing safety, connectivity, and user-preference constraints across street and trail networks.
  \item Designed and deployed an incentivized stated preference survey to quantify cyclist and micromobility rider preferences. Developed a Bayesian latent class ordered logistic model to estimate heterogeneous ridership responses to map design conditions.
  \item Developed a multi-filter bot detection pipeline (browser fingerprinting, timing analysis, XGBoost) for survey data quality control. Methodology published in \textit{Data Science for Transportation}.
\end{itemize}

\vspace{4pt}
\textbf{Regional Emissions Scenario Discovery (Connecticut)}
\begin{itemize}
  \item Contributed to a scenario discovery framework for robust regional emissions mitigation planning. Results published in \textit{Environmental Research Communications} (2026).
  \item Built and deployed a public-facing GHG inventory dashboard for Connecticut Councils of Governments (\href{https://ctghginventory.narslab.org}{ctghginventory.narslab.org}) using Apache Superset connected to a SQL database on a Linux server, hosted on AWS Lightsail.
\end{itemize}
