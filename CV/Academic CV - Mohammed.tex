\section{RESEARCH EXPERIENCE}
% ============================================================

\entry{Graduate Research Assistant}{Jan 2021 -- Present}
      {University of Massachusetts Amherst}{Amherst, MA}

\vspace{4pt}
\textbf{Schematic Bicycle Map Design and Micromobility Behavior}
\begin{itemize}
  \item Conducted a national landscape study of 684 US urban counties to characterize schematic bicycle map adoption using Bayesian mixture modeling, XGBoost, and SHAP analysis, identifying three distinct county typologies.
  \item Designed and deployed an incentivized online survey assessing cyclist and micromobility rider preferences for map design features: simplification level, geographic accuracy, infrastructure representation, and route legibility.
  \item Developed a Bayesian latent class ordered logistic regression framework to identify heterogeneous rider groups and model stated intentions to use micromobility under different map design conditions.
  \item Built a mixed-integer linear programming (MILP) optimization framework to select optimal bicycle map routes from street and trail networks, incorporating safety, connectivity, and user-preference constraints.
  \item Developed a multi-filter bot detection pipeline (platform signals, browser fingerprinting, timing analysis, XGBoost validation) for online survey quality control.
  \item \textit{Funding: Armstrong Fund for Science, 2024--2026 (PI: J.~Oke); NSF REU, 2023--2025.}
\end{itemize}

\vspace{4pt}
\textbf{Spatial Crash Typology and Risk Prediction}
\begin{itemize}
  \item Applied UMAP and Gaussian mixture models to classify 2,480 New England census tracts into crash typologies using one year of state-level crash records.
  \item Identified spatial patterns in crash frequency and severity linked to roadway geometry, land use, and socioeconomic factors across urban, suburban, and rural profiles.
  \item Built XGBoost predictive models with SHAP-based interpretation to forecast crash risk by typology, surfacing actionable feature importance for targeted safety interventions.
  \item \textit{Funding: New England University Transportation Center (NEUTC), 2024--2025 (PI: J.~Oke).}
\end{itemize}

\vspace{4pt}
\textbf{Transit OD Inference from Partial Mobile Ticketing Data}
\begin{itemize}
  \item Built a machine learning framework to infer complete origin-destination-transfer matrices from boarding-only mobile ticketing data in a regional bus network with 40+ routes.
  \item Developed passenger typology clustering using hierarchical methods and dynamic time warping to capture behavioral heterogeneity across 40,000+ weekly riders.
  \item Integrated gradient boosting spatial error correction, reducing MAE by 70\% and SMAPE by 85\% vs.\ baseline trip chaining models.
  \item \textit{Funding: Pioneer Valley Transit Authority (PVTA), 2021--2023 (PI: J.~Oke).}
\end{itemize}
